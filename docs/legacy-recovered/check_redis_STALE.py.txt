import os
import asyncio
import redis.asyncio as redis
from dotenv import load_dotenv

load_dotenv()

async def main():
    try:
        url = os.getenv("REDIS_URL", "")
        print(f"Connecting to {url}")
        r = redis.from_url(url, decode_responses=True, ssl_cert_reqs=None)
        
        # In voiceai, users might be stored with prefix like 'auth:users:*' or similar. 
        # Let's just find anything with 'user' or 'auth'
        keys = await r.keys("*auth*") + await r.keys("*user*")
        keys = list(set(keys))
        
        print(f"Keys found: {keys}")
        
        users = []
        for k in keys:
            t = await r.type(k)
            if t == "hash":
                users.append(await r.hgetall(k))
            elif t == "string":
                users.append(await r.get(k))
                
        print(f"Found {len(users)} user/auth related keys.")
        for u in users:
            print(u)
    except Exception as e:
        print(f"Error: {e}")

asyncio.run(main())
